自動 Push 排程（需管理員 PowerShell 執行一次）
powershell
$p = "C:\Users\f2289\Arc\AI_Code\leetcode-note"
$a = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-NonInteractive -WindowStyle Hidden -File `"$p\scripts\auto_push.ps1`""
Register-ScheduledTask -TaskName "LeetCode-AutoPush-2200" -Action $a -Trigger (New-ScheduledTaskTrigger -Daily -At "22:00") -RunLevel Highest -Force
Register-ScheduledTask -TaskName "LeetCode-AutoPush-0000" -Action $a -Trigger (New-ScheduledTaskTrigger -Daily -At "00:00") -RunLevel Highest -Force
